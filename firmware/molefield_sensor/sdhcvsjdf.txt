#include <WiFi.h>
#include <WiFiUdp.h>

// ================================================================
// CONFIGURATION
// ================================================================
#define BOX_ID 1 // Change to 2 when flashing the second board

const char* WIFI_SSID = "Dylan&CO."; 
const char* WIFI_PASS = "32694209"; 

// UDP Broadcast address to reach any PC on the network
const IPAddress PC_IP(255, 255, 255, 255); 
const int UDP_PORT = 5000;

// ================================================================
// HARDWARE PIN ASSIGNMENTS (HC-SR04 Sensors)
// ================================================================
#define TRIG_PIN_1 32
#define ECHO_PIN_1 35

#define TRIG_PIN_2 26
#define ECHO_PIN_2 27

// ================================================================
// GLOBAL OBJECTS & TIMING
// ================================================================
WiFiUDP udp;
unsigned long lastSendTime = 0;
const int SEND_INTERVAL_MS = 60; // 60ms allows echo pulses to settle cleanly

// ================================================================
// ULTRASONIC READING FUNCTION
// ================================================================
float readUltrasonicDistance(int trigPin, int echoPin) {
    // Clean state reset
    digitalWrite(trigPin, LOW);
    delayMicroseconds(2);
    
    // Trigger 10us pulse
    digitalWrite(trigPin, HIGH);
    delayMicroseconds(10);
    digitalWrite(trigPin, LOW);
    
    // Measure echo pulse (30,000us timeout prevents loop hanging if out of range)
    long duration_us = pulseIn(echoPin, HIGH, 30000);
    
    if (duration_us == 0) {
        return -1.0; // Return -1 if reading timed out or out of range
    }
    
    // Calculate distance in cm (Speed of sound = 0.0343 cm/us / 2)
    return duration_us * 0.01715;
}

// ================================================================
// SETUP
// ================================================================
void setup() {
    Serial.begin(115200);

    // Configure GPIO modes
    pinMode(TRIG_PIN_1, OUTPUT);
    pinMode(ECHO_PIN_1, INPUT);
    pinMode(TRIG_PIN_2, OUTPUT);
    pinMode(ECHO_PIN_2, INPUT);

    // Connect to Wi-Fi
    Serial.printf("\n[Box %d] Connecting to %s...", BOX_ID, WIFI_SSID);
    WiFi.begin(WIFI_SSID, WIFI_PASS);
    
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    
    Serial.println("\n[Wi-Fi] Connected!");
    Serial.print("[Wi-Fi] Assigned ESP32 IP: ");
    Serial.println(WiFi.localIP());

    // Initialize UDP socket
    udp.begin(UDP_PORT);
    Serial.printf("[UDP] Listening and broadcasting on port %d\n", UDP_PORT);
}

// ================================================================
// MAIN LOOP
// ================================================================
void loop() {
    if (millis() - lastSendTime >= SEND_INTERVAL_MS) {
        lastSendTime = millis();

        // 1. Read physical sensors
        float dist1 = readUltrasonicDistance(TRIG_PIN_1, ECHO_PIN_1);
        delayMicroseconds(500); // Small pause to prevent acoustic crosstalk between sensors
        float dist2 = readUltrasonicDistance(TRIG_PIN_2, ECHO_PIN_2);

        // 2. Format payload: "Box:1,S1:105.2,S2:98.4"
        char payload[64];
        snprintf(payload, sizeof(payload), "Box:%d,S1:%.1f,S2:%.1f", BOX_ID, dist1, dist2);

        // 3. Transmit packet over UDP Broadcast
        if (udp.beginPacket(PC_IP, UDP_PORT)) {
            udp.print(payload);
            
            if (udp.endPacket() == 1) {
                Serial.printf("[SUCCESS] Sent: %s\n", payload);
            } else {
                Serial.println("[ERROR] Packet failed to transmit over Wi-Fi!");
            }
        } else {
            Serial.println("[ERROR] Failed to open UDP socket!");
        }
    }
}
